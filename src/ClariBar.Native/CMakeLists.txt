cmake_minimum_required(VERSION 3.25)
project(ClariBarNative LANGUAGES CXX)
add_library(ClariBar.Native SHARED native.cpp explorer_bridge.cpp accessibility.cpp transparent_window.cpp live_picker.cpp copy_translation.cpp capture_cover.cpp crash_diagnostics.cpp)
target_compile_features(ClariBar.Native PRIVATE cxx_std_20)
target_compile_options(ClariBar.Native PRIVATE /utf-8 /W4)
target_compile_definitions(ClariBar.Native PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ClariBar.Native PRIVATE user32 gdi32 shell32 dwmapi ole32 oleaut32 dwrite comctl32 msimg32 dbghelp)
set_target_properties(ClariBar.Native PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(../ClariBar.ExplorerTap ${CMAKE_BINARY_DIR}/ExplorerTap)

add_library(ClariBar.SystemMetrics SHARED system_metrics.cpp)
target_compile_features(ClariBar.SystemMetrics PRIVATE cxx_std_20)
target_compile_options(ClariBar.SystemMetrics PRIVATE /utf-8 /W4)
target_compile_definitions(ClariBar.SystemMetrics PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ClariBar.SystemMetrics PRIVATE iphlpapi ws2_32 ole32)
set_target_properties(ClariBar.SystemMetrics PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(ClariBar.Maintenance SHARED maintenance.cpp)
target_compile_features(ClariBar.Maintenance PRIVATE cxx_std_17)
target_compile_options(ClariBar.Maintenance PRIVATE /utf-8 /W4)
target_compile_definitions(ClariBar.Maintenance PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(ClariBar.Maintenance PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(ClariBar.CopyProbe WIN32 ../../tests/ClariBar.CopyProbe/main.cpp)
target_compile_features(ClariBar.CopyProbe PRIVATE cxx_std_17)
target_compile_definitions(ClariBar.CopyProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ClariBar.CopyProbe PRIVATE user32)
set_target_properties(ClariBar.CopyProbe PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(ClariBar.TaskbarLayoutProbe ../../tests/ClariBar.TaskbarLayoutProbe/main.cpp)
target_compile_features(ClariBar.TaskbarLayoutProbe PRIVATE cxx_std_20)
target_compile_options(ClariBar.TaskbarLayoutProbe PRIVATE /utf-8 /W4 /permissive-)
target_compile_definitions(ClariBar.TaskbarLayoutProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ClariBar.TaskbarLayoutProbe PRIVATE user32 ole32 windowsapp)
target_link_options(ClariBar.TaskbarLayoutProbe PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/../../tests/ClariBar.TaskbarLayoutProbe/probe.manifest")
set_target_properties(ClariBar.TaskbarLayoutProbe PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_test(NAME taskbar-layout-compatibility COMMAND ClariBar.TaskbarLayoutProbe)
