cmake_minimum_required(VERSION 3.25)
project(ClariBarNative LANGUAGES CXX)
add_library(ClariBar.Native SHARED native.cpp explorer_bridge.cpp accessibility.cpp transparent_window.cpp live_picker.cpp)
target_compile_features(ClariBar.Native PRIVATE cxx_std_20)
target_compile_options(ClariBar.Native PRIVATE /utf-8 /W4)
target_compile_definitions(ClariBar.Native PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ClariBar.Native PRIVATE user32 gdi32 shell32 dwmapi ole32 oleaut32 dwrite comctl32)
set_target_properties(ClariBar.Native PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(../ClariBar.ExplorerTap ${CMAKE_BINARY_DIR}/ExplorerTap)

add_library(ClariBar.SystemMetrics SHARED system_metrics.cpp)
target_compile_features(ClariBar.SystemMetrics PRIVATE cxx_std_20)
target_compile_options(ClariBar.SystemMetrics PRIVATE /utf-8 /W4)
target_compile_definitions(ClariBar.SystemMetrics PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ClariBar.SystemMetrics PRIVATE iphlpapi ws2_32 ole32)
set_target_properties(ClariBar.SystemMetrics PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(ClariBar.Maintenance SHARED maintenance.cpp)
target_compile_features(ClariBar.Maintenance PRIVATE cxx_std_17)
target_compile_options(ClariBar.Maintenance PRIVATE /utf-8 /W4)
target_compile_definitions(ClariBar.Maintenance PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(ClariBar.Maintenance PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
