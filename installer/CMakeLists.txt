cmake_minimum_required(VERSION 3.25)
project(ClariBarSetup LANGUAGES CXX RC)
if(NOT EXISTS "${INNER_SETUP}")
  message(FATAL_ERROR "Compile the inner installer before building its bootstrap.")
endif()
file(TO_CMAKE_PATH "${INNER_SETUP}" INNER_SETUP)
get_filename_component(SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(BRAND_ICON "${SOURCE_ROOT}/docs/ClariBar.ico")
set(BOOTSTRAP_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/bootstrap.manifest")
configure_file(bootstrap.rc.in "${CMAKE_CURRENT_BINARY_DIR}/bootstrap.rc" @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/bootstrap.rc" PROPERTIES
  OBJECT_DEPENDS "${INNER_SETUP};${BRAND_ICON}")
add_executable(ClariBar.Setup WIN32 bootstrap.cpp "${CMAKE_CURRENT_BINARY_DIR}/bootstrap.rc")
target_compile_features(ClariBar.Setup PRIVATE cxx_std_17)
target_compile_options(ClariBar.Setup PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4>)
target_compile_definitions(ClariBar.Setup PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ClariBar.Setup PRIVATE user32 ole32)
target_link_options(ClariBar.Setup PRIVATE /MANIFEST:NO)
set_target_properties(ClariBar.Setup PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
